# ==================== MIDDLEWARE ====================
class SubscriptionMiddleware(BaseMiddleware):
    async def __call__(self, handler: Callable[[TelegramObject, Dict[str, Any]], Awaitable[Any]], event: TelegramObject, data: Dict[str, Any]) -> Any:
        bot = data.get("bot")
        user = None
        if isinstance(event, Message):
            user = event.from_user
            if event.text and event.text.startswith("/start"):
                return await handler(event, data)
        elif isinstance(event, CallbackQuery):
            user = event.from_user
            if event.data == "check_sub":
                return await handler(event, data)
        if not user or not bot:
            return await handler(event, data)

class SubscriptionMiddleware(BaseMiddleware):
    async def __call__(self, handler: Callable[[TelegramObject, Dict[str, Any]], Awaitable[Any]], event: TelegramObject, data: Dict[str, Any]) -> Any:
        bot = data.get("bot")
        user = None
        if isinstance(event, Message):
            user = event.from_user
            if event.text and event.text.startswith("/start"):
                return await handler(event, data)
        elif isinstance(event, CallbackQuery):
            user = event.from_user
            if event.data == "check_sub":
                return await handler(event, data)
        if not user or not bot:
            return await handler(event, data)
        if user.id in ADMINS:
            return await handler(event, data)
        try:
            member = await bot.get_chat_member(chat_id=CHANNEL_ID, user_id=user.id)
            if member.status in (ChatMemberStatus.MEMBER, ChatMemberStatus.ADMINISTRATOR, ChatMemberStatus.CREATOR):
                return await handler(event, data)
        except Exception:
            return await handler(event, data)
        text = "⚠️ <b>Botdan foydalanish uchun kanalimizga obuna bo'lishingiz shart!</b>\n\nQuyidagi tugma orqali obuna bo'ling, keyin «Obuna bo'ldim» ni bosing."
        if isinstance(event, Message):
            await event.answer(text, reply_markup=subscription_kb(), parse_mode="HTML")
        elif isinstance(event, CallbackQuery):
            await event.answer("Avval kanalga obuna bo'ling!", show_alert=True)
            await event.message.answer(text, reply_markup=subscription_kb(), parse_mode="HTML")
        return

@router.callback_query(F.data == "quiz_start")
async def quiz_start(callback: CallbackQuery, state: FSMContext):
    await state.set_state(QuizState.answering)
    await state.update_data(current=0, score=0)
    await send_question(callback, state)
    await callback.answer()


async def send_question(callback: CallbackQuery, state: FSMContext):
    data = await state.get_data()
    idx = data.get("current", 0)
    if idx >= len(QUIZ_QUESTIONS):
        score = data.get("score", 0)
        total = len(QUIZ_QUESTIONS)
        percent = int(score / total * 100)
        text = f"🏁 <b>Quiz tugadi!</b>\n\nTo'g'ri: <b>{score}/{total}</b> ({percent}%)\n\n"
        if percent >= 80:
            text += "🎉 Ajoyib natija!"
        elif percent >= 50:
            text += "👍 Yaxshi, yana o'qing."
        else:
            text += "📚 Darslarni qayta o'qing."
        await state.clear()
        await callback.message.edit_text(text, parse_mode="HTML")
        await callback.message.answer("🏠 Bosh menyu:", reply_markup=main_menu_kb())
        return
    q = QUIZ_QUESTIONS[idx]
    await callback.message.edit_text(
        f"❓ <b>Savol {idx+1}/{len(QUIZ_QUESTIONS)}</b>\n\n{q['q']}",
        reply_markup=quiz_answer_kb(idx, q["options"]), parse_mode="HTML"
    )


@router.callback_query(F.data.startswith("quiz_ans:"), QuizState.answering)
async def quiz_answer(callback: CallbackQuery, state: FSMContext):
    parts = callback.data.split(":")
    q_id, ans_id = int(parts[1]), int(parts[2])
    data = await state.get_data()
    current = data.get("current", 0)
    score = data.get("score", 0)
    if q_id != current:
        await callback.answer("Bu savol allaqachon javob berilgan", show_alert=True)
        return
    if ans_id == QUIZ_QUESTIONS[q_id]["correct"]:
        score += 1
        await callback.answer("✅ To'g'ri!")
    else:
        await callback.answer("❌ Noto'g'ri")
    await state.update_data(current=current + 1, score=score)
    await send_question(callback, state)

    await dp.start_polling(bot)
